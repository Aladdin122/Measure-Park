def extract_number_from_text(text):
    matches = re.findall(r"\d+\.\d+|\d+", text)
    if matches:
        return float(matches[0])
    return None

def get_ocr_value(roi, reader):
    if roi is None or roi.size == 0:
        return None
    try:
        roi_large = cv2.resize(roi, (0, 0), fx=2.0, fy=2.0, interpolation=cv2.INTER_CUBIC)
        gray = cv2.cvtColor(roi_large, cv2.COLOR_BGR2GRAY)
        thresh = cv2.adaptiveThreshold(gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2)

        results = reader.readtext(roi_large)
        if not results:
            results = reader.readtext(thresh)

        if results:
            for res in results:
                val = extract_number_from_text(res[1])
                if val is not None:
                    return val
    except Exception as e:
        print(f"[WARN] OCR pre-processing failure: {e}")
    return None

def detect_wheel_center_sam3_fit_ellipse(park_cv, roi_bounds, model, processor, device):
    roi_xmin, roi_xmax, roi_ymin, roi_ymax = roi_bounds
    roi_cv = park_cv[roi_ymin:roi_ymax, roi_xmin:roi_xmax]
    if roi_cv.size == 0:
        return None

    try:
        roi_pil = Image.fromarray(cv2.cvtColor(roi_cv, cv2.COLOR_BGR2RGB))
        inputs_wheel = processor(images=roi_pil, text="car wheel", return_tensors="pt").to(device)

        with torch.no_grad():
            outputs_wheel = model(**inputs_wheel)

        results_wheel = processor.post_process_instance_segmentation(
            outputs_wheel, threshold=0.25, mask_threshold=0.3, target_sizes=inputs_wheel.get("original_sizes").tolist()
        )[0]

        if len(results_wheel["masks"]) == 0:
            return None

        best_candidate = None
        best_score = -1.0

        for mask_tensor in results_wheel["masks"]:
            mask_np = mask_tensor.cpu().numpy().astype(np.uint8) * 255
            contours, _ = cv2.findContours(mask_np, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
            if not contours:
                continue

            largest_cnt = max(contours, key=cv2.contourArea)
            area = cv2.contourArea(largest_cnt)

            if area < 80 or len(largest_cnt) < 5:
                continue

            (ellipse_cx, ellipse_cy), (d1, d2), angle = cv2.fitEllipse(largest_cnt)

            gx = roi_xmin + int(round(ellipse_cx))
            gy = roi_ymin + int(round(ellipse_cy))

            if gx < 0 or gx >= park_cv.shape[1] or gy < 0 or gy >= park_cv.shape[0]:
                continue

            axis_ratio = min(d1, d2) / max(1e-5, max(d1, d2))
            if axis_ratio < 0.40:
                continue

            score = area * axis_ratio
            if score > best_score:
                best_score = score
                best_candidate = (gx, gy, 0.95)

        return best_candidate
    except Exception as e:
        print(f"[WARN] SAM 3 wheel fitEllipse error: {e}")
        return None

def detect_single_wheel_hub_robust(roi_gray, expected_x, expected_y, roi_xmin, roi_ymin, ch):
    if roi_gray.size == 0:
        return expected_x, expected_y, 0.20

    roi_blur = cv2.GaussianBlur(roi_gray, (7, 7), 0)
    min_r = max(8, int(ch * 0.08))
    max_r = int(ch * 0.24)

    for p2 in [22, 18, 14, 10]:
        circles = cv2.HoughCircles(
            roi_blur, cv2.HOUGH_GRADIENT, dp=1.0, minDist=12,
            param1=45, param2=p2, minRadius=min_r, maxRadius=max_r
        )
        if circles is not None:
            circles = np.round(circles[0]).astype(int)
            candidates = []
            for local_x, local_y, r in circles:
                gx = roi_xmin + local_x
                gy = roi_ymin + local_y
                x_err = abs(gx - expected_x)
                if x_err <= int(ch * 0.35):
                    candidates.append({"x": gx, "y": gy, "r": r, "err": x_err})

            if candidates:
                best_cand = min(candidates, key=lambda c: c["err"])
                cluster = [c for c in candidates if abs(c["x"] - best_cand["x"]) <= int(ch * 0.06)]
                hub_x = int(np.median([c["x"] for c in cluster]))
                hub_y = int(np.median([c["y"] for c in cluster]))
                return hub_x, hub_y, 0.85

    return expected_x, expected_y, 0.20

def compute_region_similarity(template_gray, template_hsv, template_mask, crop_gray, crop_hsv, crop_mask, sift, bf):
    if template_gray is None or crop_gray is None or template_gray.size == 0 or crop_gray.size == 0:
        return 0.0

    th, tw = template_gray.shape[:2]
    crop_gray_resized = cv2.resize(crop_gray, (tw, th), interpolation=cv2.INTER_CUBIC)
    crop_hsv_resized = cv2.resize(crop_hsv, (tw, th), interpolation=cv2.INTER_CUBIC)
    crop_mask_resized = cv2.resize(crop_mask, (tw, th), interpolation=cv2.INTER_NEAREST)

    kp_tmpl, des_tmpl = sift.detectAndCompute(template_gray, None)
    kp_crop, des_crop = sift.detectAndCompute(crop_gray, None)

    good_matches = []
    if des_tmpl is not None and des_crop is not None and len(des_crop) > 1:
        matches = bf.knnMatch(des_tmpl, des_crop, k=2)
        for m_list in matches:
            if len(m_list) == 2:
                m, n = m_list
                if m.distance < 0.75 * n.distance:
                    good_matches.append(m)

    sift_score = len(good_matches)
    hist_tmpl = cv2.calcHist([template_hsv], [0, 1], template_mask, [50, 60], [0, 180, 0, 256])
    hist_crop = cv2.calcHist([crop_hsv_resized], [0, 1], crop_mask_resized, [50, 60], [0, 180, 0, 256])
    cv2.normalize(hist_tmpl, hist_tmpl, 0, 1, cv2.NORM_MINMAX)
    cv2.normalize(hist_crop, hist_crop, 0, 1, cv2.NORM_MINMAX)
    color_corr = cv2.compareHist(hist_tmpl, hist_crop, cv2.HISTCMP_CORREL)

    shape_match = cv2.matchShapes(template_mask, crop_mask_resized, cv2.CONTOURS_MATCH_I1, 0.0)
    shape_score = 1.0 / (1.0 + shape_match)

    ar_tmpl = tw / max(1, th)
    ar_crop = crop_gray.shape[1] / max(1, crop_gray.shape[0])
    aspect_score = min(ar_tmpl, ar_crop) / max(ar_tmpl, ar_crop)

    return (sift_score * 3.0) + (color_corr * 80.0) + (shape_score * 40.0) + (aspect_score * 30.0)

def detect_overhang_semantic_features(park_cv, crop_bounds, model, processor, device):
    xmin, xmax, ymin, ymax = crop_bounds
    crop = park_cv[ymin:ymax, xmin:xmax]
    if crop.size == 0:
        return False, False, None

    crop_pil = Image.fromarray(cv2.cvtColor(crop, cv2.COLOR_BGR2RGB))
    front_prompts = ["headlight", "front grille"]
    rear_prompts = ["taillight", "rear light"]

    def query_sam3_prompts(prompts):
        for prompt in prompts:
            try:
                inputs = processor(images=crop_pil, text=prompt, return_tensors="pt").to(device)
                with torch.no_grad():
                    outputs = model(**inputs)
                results = processor.post_process_instance_segmentation(
                    outputs, threshold=0.25, mask_threshold=0.3, target_sizes=inputs.get("original_sizes").tolist()
                )[0]
                if len(results["masks"]) > 0:
                    for mask_tensor in results["masks"]:
                        area = torch.sum(mask_tensor).item()
                        if area > 40:
                            return True, prompt
            except Exception as e:
                pass
        return False, None

    front_detected, front_feature = query_sam3_prompts(front_prompts)
    rear_detected, rear_feature = query_sam3_prompts(rear_prompts)
    detected_feature = front_feature if front_detected else rear_feature
    return front_detected, rear_detected, detected_feature

def execute_three_image_analysis_colab(ref_original_path="chery_ref.jpg", ref_annotated_path="nextout.jpg", park_image_path="park4.jpeg", output_path="outsample2.jpg"):
    t_start = time.time()
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"[INFO] Launching Stage 2 on device: {device}")
    if device == "cuda":
        print(f"[INFO] Active GPU: {torch.cuda.get_device_name(0)}")

    annotated_img = cv2.imread(ref_annotated_path)
    if annotated_img is None:
        print(f"[ERROR] Could not load: {ref_annotated_path}")
        return

    ref_hsv = cv2.cvtColor(annotated_img, cv2.COLOR_BGR2HSV)
    ref_h, ref_w, _ = annotated_img.shape

    lower_orange = np.array([5, 100, 100])
    upper_orange = np.array([22, 255, 255])
    orange_mask = cv2.inRange(ref_hsv, lower_orange, upper_orange)
    orange_pixels = np.argwhere(orange_mask > 0)

    lower_lime = np.array([35, 100, 100])
    upper_lime = np.array([55, 255, 255])
    lime_mask = cv2.inRange(ref_hsv, lower_lime, upper_lime)
    lime_pixels = np.argwhere(lime_mask > 0)

    total_car_length_cm, orange_val_cm, green_val_cm = None, None, None

    if EASYOCR_AVAILABLE and len(orange_pixels) > 0 and len(lime_pixels) > 0:
        try:
            reader = easyocr.Reader(['en'], gpu=torch.cuda.is_available())
            oy_min, ox_min, ox_max = np.min(orange_pixels[:, 0]), np.min(orange_pixels[:, 1]), np.max(orange_pixels[:, 1])
            roi_orange = annotated_img[max(0, oy_min - 45):oy_min + 10, max(0, ox_min - 20):ox_max + 20]

            gy_min, gx_min, gx_max = np.min(lime_pixels[:, 0]), np.min(lime_pixels[:, 1]), np.max(lime_pixels[:, 1])
            roi_lime = annotated_img[max(0, gy_min - 45):gy_min + 10, max(0, gx_min - 20):gx_max + 20]
            roi_ruler = annotated_img[0:int(ref_h * 0.35), int(ref_w * 0.25):int(ref_w * 0.75)]

            orange_val_cm = get_ocr_value(roi_orange, reader)
            green_val_cm = get_ocr_value(roi_lime, reader)
            total_car_length_cm = get_ocr_value(roi_ruler, reader)
        except Exception as e:
            print(f"[WARN] OCR parse fallback: {e}")

    ox_min_val, ox_max_val = np.min(orange_pixels[:, 1]), np.max(orange_pixels[:, 1])
    gx_min_val, gx_max_val = np.min(lime_pixels[:, 1]), np.max(lime_pixels[:, 1])
    car_span_xmin = min(ox_min_val, gx_min_val)
    car_span_xmax = max(ox_max_val, gx_max_val)
    span_w = car_span_xmax - car_span_xmin

    if total_car_length_cm is None: total_car_length_cm = 461.0
    ref_pixels_per_cm = span_w / total_car_length_cm
    if orange_val_cm is None: orange_val_cm = (ox_max_val - ox_min_val) / ref_pixels_per_cm
    if green_val_cm is None: green_val_cm = (gx_max_val - gx_min_val) / ref_pixels_per_cm

    front_overhang_cm, rear_overhang_cm = orange_val_cm, green_val_cm
    orange_rel_x1 = (ox_min_val - car_span_xmin) / span_w
    orange_rel_x2 = (ox_max_val - car_span_xmin) / span_w
    green_rel_x1 = (gx_min_val - car_span_xmin) / span_w
    green_rel_x2 = (gx_max_val - car_span_xmin) / span_w

    original_ref_pil = Image.open(ref_original_path).convert("RGB")
    original_ref_cv = cv2.imread(ref_original_path)

    # Model initialization
    model = Sam3Model.from_pretrained("facebook/sam3").to(device)
    processor = Sam3Processor.from_pretrained("facebook/sam3")

    # Segment clean template
    inputs_ref = processor(images=original_ref_pil, text="car", return_tensors="pt").to(device)
    with torch.no_grad():
        outputs_ref = model(**inputs_ref)
    results_ref = processor.post_process_instance_segmentation(
        outputs_ref, threshold=0.3, mask_threshold=0.4, target_sizes=inputs_ref.get("original_sizes").tolist()
    )[0]

    mask_areas_ref = [torch.sum(m).item() for m in results_ref["masks"]]
    clean_ref_mask = results_ref["masks"][np.argmax(mask_areas_ref)].cpu().numpy().astype(np.uint8) * 255
    rx_ref, ry_ref, rw_ref, rh_ref = cv2.boundingRect(cv2.findNonZero(clean_ref_mask))

    ref_gray = cv2.cvtColor(original_ref_cv, cv2.COLOR_BGR2GRAY)
    ref_hsv = cv2.cvtColor(original_ref_cv, cv2.COLOR_BGR2HSV)

    ox1, ox2 = rx_ref + int(orange_rel_x1 * rw_ref), rx_ref + int(orange_rel_x2 * rw_ref)
    tmpl_orange_gray = ref_gray[ry_ref:ry_ref+rh_ref, ox1:ox2]
    tmpl_orange_hsv = ref_hsv[ry_ref:ry_ref+rh_ref, ox1:ox2]
    tmpl_orange_mask = clean_ref_mask[ry_ref:ry_ref+rh_ref, ox1:ox2]

    gx1, gx2 = rx_ref + int(green_rel_x1 * rw_ref), rx_ref + int(green_rel_x2 * rw_ref)
    tmpl_green_gray = ref_gray[ry_ref:ry_ref+rh_ref, gx1:gx2]
    tmpl_green_hsv = ref_hsv[ry_ref:ry_ref+rh_ref, gx1:gx2]
    tmpl_green_mask = clean_ref_mask[ry_ref:ry_ref+rh_ref, gx1:gx2]

    # Live parking scene
    park_pil = Image.open(park_image_path).convert("RGB")
    park_cv = cv2.imread(park_image_path)
    park_h, park_w, _ = park_cv.shape

    t_scene_start = time.time()
    inputs_park = processor(images=park_pil, text="car", return_tensors="pt").to(device)
    with torch.no_grad():
        outputs_park = model(**inputs_park)
    results_park = processor.post_process_instance_segmentation(
        outputs_park, threshold=0.3, mask_threshold=0.4, target_sizes=inputs_park.get("original_sizes").tolist()
    )[0]

    mask_areas_p = [torch.sum(m).item() for m in results_park["masks"]]
    sorted_idx_p = np.argsort(mask_areas_p)[::-1]
    mask_a = results_park["masks"][sorted_idx_p[0]].cpu().numpy().astype(np.uint8) * 255
    mask_b = results_park["masks"][sorted_idx_p[1]].cpu().numpy().astype(np.uint8) * 255

    coords_a, coords_b = cv2.boundingRect(cv2.findNonZero(mask_a)), cv2.boundingRect(cv2.findNonZero(mask_b))
    if (coords_a[0] + coords_a[2] // 2) < (coords_b[0] + coords_b[2] // 2):
        left_mask, left_coords, right_mask, right_coords = mask_a, coords_a, mask_b, coords_b
    else:
        left_mask, left_coords, right_mask, right_coords = mask_b, coords_b, mask_a, coords_a

    gray_park = cv2.cvtColor(park_cv, cv2.COLOR_BGR2GRAY)
    sift = cv2.SIFT_create()
    bf = cv2.BFMatcher()

    lx, ly, lw, lh = left_coords
    rx, ry, rw, rh = right_coords
    ref_gray_crop = ref_gray[ry_ref:ry_ref+rh_ref, rx_ref:rx_ref+rw_ref]
    ref_hsv_crop = ref_hsv[ry_ref:ry_ref+rh_ref, rx_ref:rx_ref+rw_ref]
    ref_mask_crop = clean_ref_mask[ry_ref:ry_ref+rh_ref, rx_ref:rx_ref+rw_ref]

    left_gray_crop = gray_park[ly:ly+lh, lx:lx+lw]
    right_gray_crop = gray_park[ry:ry+rh, rx:rx+rw]
    left_hsv_crop = cv2.cvtColor(park_cv[ly:ly+lh, lx:lx+lw], cv2.COLOR_BGR2HSV)
    right_hsv_crop = cv2.cvtColor(park_cv[ry:ry+rh, rx:rx+rw], cv2.COLOR_BGR2HSV)

    match_left = compute_region_similarity(ref_gray_crop, ref_hsv_crop, ref_mask_crop, left_gray_crop, left_hsv_crop, left_mask[ly:ly+lh, lx:lx+lw], sift, bf)
    match_right = compute_region_similarity(ref_gray_crop, ref_hsv_crop, ref_mask_crop, right_gray_crop, right_hsv_crop, right_mask[ry:ry+rh, rx:rx+rw], sift, bf)

    if match_left > match_right:
        target_pos, target_mask, target_box, other_mask, other_box = "left", left_mask, left_coords, right_mask, right_coords
        gap_side = "right"
    else:
        target_pos, target_mask, target_box, other_mask, other_box = "right", right_mask, right_coords, left_mask, left_coords
        gap_side = "left"

    cx, cy, cw, ch = target_box
    ox, oy, ow, oh = other_box
    y_indices, _ = np.where(target_mask > 0)
    y_bottom = np.max(y_indices) if len(y_indices) > 0 else (cy + ch)

    expected_wheel_x = cx + cw - int(ch * 0.425) if gap_side == "right" else cx + int(ch * 0.425)
    roi_bounds = (max(cx, expected_wheel_x - int(ch * 0.45)), min(cx + cw, expected_wheel_x + int(ch * 0.45)), max(cy, int(cy + ch * 0.48)), min(park_h, y_bottom + 10))

    sam_wheel_res = detect_wheel_center_sam3_fit_ellipse(park_cv, roi_bounds, model, processor, device)
    if sam_wheel_res:
        active_wheel_x, active_wheel_y, wheel_conf = sam_wheel_res
    else:
        active_wheel_x, active_wheel_y, wheel_conf = detect_single_wheel_hub_robust(gray_park[roi_bounds[2]:roi_bounds[3], roi_bounds[0]:roi_bounds[1]], expected_wheel_x, int(cy + ch * 0.78), roi_bounds[0], roi_bounds[2], ch)

    ref_pixels_x = np.where(target_mask > 0)[1]
    other_pixels_x = np.where(other_mask > 0)[1]

    if gap_side == "left":
        ref_bumper_x, obstacle_bumper_x = int(np.min(ref_pixels_x)), int(np.max(other_pixels_x))
        overhang_xmin, overhang_xmax = ref_bumper_x, active_wheel_x
    else:
        ref_bumper_x, obstacle_bumper_x = int(np.max(ref_pixels_x)), int(np.min(other_pixels_x))
        overhang_xmin, overhang_xmax = active_wheel_x, ref_bumper_x

    front_det, rear_det, det_feat_name = detect_overhang_semantic_features(park_cv, (overhang_xmin, overhang_xmax, cy, cy + ch), model, processor, device)

    if front_det and not rear_det:
        selected_overhang_cm, overhang_type = orange_val_cm, "front"
    elif rear_det and not front_det:
        selected_overhang_cm, overhang_type = green_val_cm, "rear"
    else:
        selected_overhang_cm, overhang_type = orange_val_cm, "front"

    overhang_px = abs(active_wheel_x - ref_bumper_x)
    px_per_cm = overhang_px / selected_overhang_cm
    gap_pixels = abs(ref_bumper_x - obstacle_bumper_x)
    dist_cm = gap_pixels / px_per_cm

    # Visualization
    bg = np.full_like(park_cv, 255)
    overlay = cv2.addWeighted(park_cv, 0.35, np.full_like(park_cv, (215, 120, 195)), 0.65, 0)
    smooth_m = np.repeat((cv2.GaussianBlur(cv2.bitwise_or(target_mask, other_mask), (5, 5), 0).astype(float) / 255.0)[:, :, np.newaxis], 3, axis=2)
    canvas = (overlay * smooth_m + bg * (1.0 - smooth_m)).astype(np.uint8)

    cv2.rectangle(canvas, (ox, oy - 15), (ox + ow, oy + oh + 15), (165, 80, 140), 2)
    cv2.rectangle(canvas, (overhang_xmin, cy - 15), (overhang_xmax, cy + ch + 15), (165, 80, 140), 2)
    cv2.circle(canvas, (active_wheel_x, active_wheel_y), 5, (0, 0, 255), -1)
    cv2.circle(canvas, (active_wheel_x, active_wheel_y), 7, (255, 255, 255), 2)

    res_label = f"Distance: {dist_cm:.1f} cm"
    (rw, rh), _ = cv2.getTextSize(res_label, cv2.FONT_HERSHEY_SIMPLEX, 0.5, 2)
    cx_box = min(ref_bumper_x, obstacle_bumper_x) + (gap_pixels // 2)
    cv2.rectangle(canvas, (cx_box - (rw // 2) - 10, 40), (cx_box + (rw // 2) + 10, 40 + rh + 14), (255, 255, 255), -1)
    cv2.putText(canvas, res_label, (cx_box - (rw // 2), 40 + rh + 7), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 0), 2, cv2.LINE_AA)

    cv2.imwrite(output_path, canvas)
    total_time = time.time() - t_start

    print("\n" + "="*45)
    print(f"STAGE 2 TOTAL RUNTIME: {total_time:.2f} seconds")
    print(f"Target Position:       {target_pos.upper()}")
    print(f"Measured Clearance:    {dist_cm:.1f} cm")
    print("="*45)

    plt.figure(figsize=(12, 7))
    plt.imshow(cv2.cvtColor(canvas, cv2.COLOR_BGR2RGB))
    plt.axis("off")
    plt.title(f"Stage 2 Result - Clearance: {dist_cm:.1f} cm (GPU Latency: {total_time:.2f}s)")
    plt.show()
